        public static double CalculateTotalPayroll(Employee[] employees)
        {
            double totalSalary = 0;
            foreach (Employee employee in employees)
            {
                if (employee != null)
                {
                    totalSalary += employee.CalculateSalary();
                }
            }
            return totalSalary;
        }
        public static Employee FindHighestPaid(Employee[] employees)
        {
            Employee highestPaid = employees[0];
            double highestSalary = 0;
            foreach (Employee employee in employees)
            {
                double salary = employee.CalculateSalary();
                if (salary > highestSalary)
                {
                    highestSalary = salary;
                    highestPaid = employee;
                }
            }
            return highestPaid;
        }
        public static void SortEmployeesBySalary(Employee[] employees)
        {
            for (int i = 0; i < employees.Length - 1; i++)
            {
                for (int j = 0; j < employees.Length - i - 1; j++)
                {
                    if (employees[j].CalculateSalary() < employees[j + 1].CalculateSalary())
                    {
                        Employee temp = employees[j];
                        employees[j] = employees[j + 1];
                        employees[j + 1] = temp;
                    }
                }
            }
        }
    }
